Conexión con tu hoja de Google

La app escribe en tus hojas Presupuesto, Movimientos e Ingresos Extra

Sin conectar — la app funciona con tus números cargados
La URL se usa solo mientras la app está abierta; no se guarda en ningún servidor.

Para conectar, una sola vez:

  1. Abre tu hoja de presupuesto y entra a Extensiones → Apps Script.
  2. Crea un archivo nuevo (o pega al final de tu Code.gs) el código puente de abajo y guarda.
  3. Arriba a la derecha: Implementar → Nueva implementación → Aplicación web. Ejecutar como: Yo. Acceso: Cualquier persona.
  4. Copia la URL que termina en /exec y pégala aquí arriba. Dale Probar conexión.
  5. Desde ese momento, cada gasto o ingreso extra que registres se envía solo a tu hoja, y con Traer movimientos ves en la app lo que ya tenías capturado en Sheets.
Código puente para Apps Script (clic para ver y copiar) 
// ===== PUENTE PRESUPUESTO SOLO =====
// Pégalo en el mismo proyecto de Apps Script de tu hoja
// (Extensiones > Apps Script) y publícalo como Aplicación web.

function doGet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var p = ss.getSheetByName('Presupuesto');
  var mov = ss.getSheetByName('Movimientos');
  var filas = [];
  if (mov && mov.getLastRow() > 3) {
    filas = mov.getRange(4, 1, mov.getLastRow() - 3, 5).getValues();
  }
  var movimientos = [];
  for (var i = 0; i < filas.length; i++) {
    var r = filas[i];
    if (!r[0]) continue;
    movimientos.push({
      fecha: Utilities.formatDate(new Date(r[0]), Session.getScriptTimeZone(), 'yyyy-MM-dd'),
      tipo: String(r[1] || 'Gasto'),
      categoria: String(r[2] || 'Otro'),
      monto: Number(r[3] || 0),
      nota: String(r[4] || '')
    });
  }
  return ContentService
    .createTextOutput(JSON.stringify({
      ok: true,
      ingresoQuincenal: p.getRange('B3').getValue(),
      ingresoMensual: p.getRange('B4').getValue(),
      totalGastos: p.getRange('E20').getValue(),
      totalAsignado: p.getRange('E27').getValue(),
      patrimonio: p.getRange('B41').getValue(),
      movimientos: movimientos
    }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var d = JSON.parse(e.postData.contents);
  var fecha = new Date(d.fecha + 'T12:00:00');
  if (d.tipo === 'Ingreso extra') {
    var h = ss.getSheetByName('Ingresos Extra');
    h.appendRow([fecha, d.nota || d.categoria, d.categoria, d.monto, d.metodo || 'Efectivo', 'Desde la app']);
  } else {
    var m = ss.getSheetByName('Movimientos');
    m.appendRow([fecha, 'Gasto', d.categoria, d.monto, d.nota || '']);
  }
  return ContentService
    .createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
